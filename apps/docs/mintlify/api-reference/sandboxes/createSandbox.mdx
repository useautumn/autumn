import { DynamicParamField } from "/snippets/dynamic-param-field.jsx";
import { DynamicResponseField } from "/snippets/dynamic-response-field.jsx";

### Body Parameters

<DynamicParamField body="name" type="string" required>
  A name for the sandbox, unique within your organization.
</DynamicParamField>

<DynamicParamField body="color" type="'gray' | 'blue' | 'green' | 'amber' | 'red' | 'pink'">
  Colour the dashboard uses to label the sandbox. Defaults to `gray`.
</DynamicParamField>

<DynamicParamField body="icon" type="string">
  Icon the dashboard uses to label the sandbox. Defaults to `Flask`.
</DynamicParamField>


### Response

<DynamicResponseField name="id" type="string">
  The sandbox's organization ID.
</DynamicResponseField>

<DynamicResponseField name="name" type="string">
  The sandbox's name.
</DynamicResponseField>

<DynamicResponseField name="slug" type="string">
  The sandbox's slug, derived from its name.
</DynamicResponseField>

<DynamicResponseField name="created_at" type="number">
  When the sandbox was created, ms since epoch.
</DynamicResponseField>

<DynamicResponseField name="color" type="'gray' | 'blue' | 'green' | 'amber' | 'red' | 'pink'">
  Colour the dashboard uses to label the sandbox.
</DynamicResponseField>

<DynamicResponseField name="icon" type="string">
  Icon the dashboard uses to label the sandbox.
</DynamicResponseField>

<DynamicResponseField name="secret_key" type="string">
  The sandbox's own secret key. Shown once, here — store it before you discard the response.
</DynamicResponseField>
