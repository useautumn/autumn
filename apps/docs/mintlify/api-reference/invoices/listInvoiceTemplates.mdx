### Body Parameters

<DynamicParamField body="offset" type="integer">
  Number of items to skip
</DynamicParamField>

<DynamicParamField body="limit" type="integer">
  Number of items to return. Default 10, max 1000.
</DynamicParamField>


### Response

<DynamicResponseField name="list" type="object[]">
  Invoice templates, newest first.
  <Expandable title="properties">
    <DynamicResponseField name="id" type="string">
      Unique identifier for the invoice template.
    </DynamicResponseField>

    <DynamicResponseField name="name" type="string">
      User-defined name to distinguish this template when sending an invoice.
    </DynamicResponseField>

    <DynamicResponseField name="footer" type="string">
      Footer text rendered on the invoice, typically bank details so customers can pay directly.
    </DynamicResponseField>

    <DynamicResponseField name="memo" type="string">
      Memo shown near the top of the invoice (Stripe invoice description), e.g. a contact line for payment questions.
    </DynamicResponseField>

    <DynamicResponseField name="net_terms_days" type="integer">
      Default number of days the customer has to pay before the invoice is due. Overridable when sending.
    </DynamicResponseField>

    <DynamicResponseField name="created_at" type="number">
      Timestamp (ms) when the template was created.
    </DynamicResponseField>

  </Expandable>
</DynamicResponseField>

<DynamicResponseField name="total" type="number">
  Number of templates returned in this page.
</DynamicResponseField>

<DynamicResponseField name="limit" type="number" />

<DynamicResponseField name="offset" type="number" />

<DynamicResponseField name="has_more" type="boolean">
  Whether more templates exist after this page.
</DynamicResponseField>
